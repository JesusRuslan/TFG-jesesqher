\begin{entrada}{2025-XX-XX}{Puesta en marcha del entorno de simulación}
  \problema{Necesidad de reproducir el comportamiento del TurtleBot4 sin disponer del robot físico en todo momento.}
  \solucion{Instalación de \texttt{ros-humble-turtlebot4-simulator} e Ignition Gazebo Fortress. Lanzamiento con:
    \texttt{ros2 launch turtlebot4\_ignition\_bringup turtlebot4\_ignition.launch.py}}
  \secciondestino{5.1 — Introducción a la implementación}
  \nota{La simulación incluye el modelo URDF completo del TurtleBot4 Lite con el Create3 y el sensor LiDAR RPLIDAR A1.}
\end{entrada}


\begin{entrada}{2025-XX-XX}{Teleop con teclado — verificación del entorno}
  \solucion{Verificación del entorno mediante \texttt{teleop\_twist\_keyboard}. El robot responde a comandos en \texttt{/cmd\_vel}.}
  \secciondestino{5.1 — Introducción a la implementación}
  \nota{Primer contacto con el topic \texttt{/cmd\_vel} (\texttt{geometry\_msgs/Twist}), que es la interfaz de control fundamental para todos los algoritmos posteriores.}
\end{entrada}


\begin{entrada}{2025-XX-XX}{Nodo C++ — control open-loop con botones del Create3}
  \problema{Mover el robot una distancia fija ($d = 0{,}5$\,m) mediante los botones físicos del Create3 (\texttt{button\_1}, \texttt{button\_2}).}
  \solucion{Nodo suscrito a \texttt{/interface\_buttons} que publica en \texttt{/cmd\_vel}. Control open-loop: la duración del movimiento se calcula como $t = d / v$.}
  \secciondestino{5.2 — Desarrollo de algoritmos (control básico)}
  \nota{Primera implementación de referencia. El error de posición final de este método será la línea base para comparar con el control closed-loop.}
\end{entrada}


\begin{entrada}{2025-XX-XX}{Problema: el robot para solo tras un único comando de velocidad}
  \problema{El robot avanzaba $\approx 0{,}15$\,m y se detenía sin que se alcanzara la distancia objetivo ni se publicara el mensaje de parada.}
  \causa{El plugin del Create3 en Ignition Gazebo implementa un \textit{safety timeout}: si deja de recibir comandos en \texttt{/cmd\_vel} con regularidad, el robot para automáticamente. Una sola publicación no es suficiente.}
  \solucion{Sustituir la publicación única por un \texttt{wall\_timer} a 10\,Hz que publica continuamente el \texttt{Twist} mientras \texttt{moving\_} sea \texttt{true}. El timer se cancela al alcanzar la distancia o al pulsar Button 2.}
\begin{lstlisting}[language=C++, caption={Timer de publicación continua en /cmd\_vel}]
cmd_vel_timer_ = this->create_wall_timer(
  100ms,
  [this]() {
    if (!moving_) { cmd_vel_timer_->cancel(); return; }
    auto twist = geometry_msgs::msg::Twist();
    twist.linear.x = LINEAR_SPEED;
    cmd_vel_publisher_->publish(twist);
  });
\end{lstlisting}
  \secciondestino{5.3 — Metodología experimental (incidencias)}
  \resultado{El comportamiento del simulador es fiel al robot real: el Create3 físico exige el mismo patrón de publicación continua a $\geq 10$\,Hz.}
\end{entrada}


\begin{entrada}{2025-XX-XX}{Nodo C++ — control closed-loop con odometría}
  \problema{El control open-loop acumula error porque asume velocidad constante, ignorando la rampa de aceleración interna del Create3.}
  \solucion{Suscripción a \texttt{/odom} (\texttt{nav\_msgs/Odometry}). Se guarda la posición de inicio $(x_0, y_0)$ al pulsar Button 1 y se para cuando:
    \[ d = \sqrt{(x - x_0)^2 + (y - y_0)^2} \geq d_{\text{objetivo}} \]}
  \secciondestino{5.2 — Desarrollo de algoritmos (control con realimentación)}
  \resultado{Comparativa open-loop vs.\ closed-loop: medir el error de posición final $\varepsilon = |d_{\text{real}} - d_{\text{objetivo}}|$ en ambos métodos. Ver §\,6.3.}
\end{entrada}


\begin{entrada}{2025-XX-XX}{Problema: velocidad saturada a $\approx 0{,}31$\,m/s}
  \problema{Al comandar \texttt{LINEAR\_SPEED = 2.0}\,m/s, la odometría reporta $\approx 0{,}31$\,m/s.}
  \causa{El iRobot Create3 impone un límite de velocidad máxima de \textbf{0{,}306\,m/s} por firmware, tanto en robot real como en simulación. Cualquier valor superior se satura.}
  \solucion{Usar \texttt{LINEAR\_SPEED = 0.3}\,m/s como velocidad nominal máxima.}
  \secciondestino{3.3 — Requisitos del sistema (restricciones hardware)}
  \nota{Este límite es un requisito hardware del sistema, no un bug. Debe documentarse en la sección de requisitos.}
\end{entrada}


\begin{entrada}{2025-XX-XX}{Problema: artefactos de compilación con colcon}
  \problema{Tras modificar las constantes \texttt{LINEAR\_SPEED} y \texttt{TARGET\_DIST}, el robot seguía comportándose con los valores antiguos (recorría 5\,m en lugar de 0{,}5\,m).}
  \causa{Los directorios \texttt{build/} e \texttt{install/} cacheaban el binario anterior. \texttt{colcon build} sin limpiar no recompiló el ejecutable.}
  \solucion{Build limpio:
\begin{lstlisting}[language=bash]
rm -rf build/ install/ log/
colcon build --packages-select turtlebot4_cpp_tutorials
source install/setup.bash
\end{lstlisting}}
  \secciondestino{5.3 — Metodología experimental (decisiones técnicas)}
  \nota{Buena práctica: ante comportamientos inesperados, verificar siempre que el binario en ejecución corresponde al código fuente actual.}
\end{entrada}


% ── SLAM Y NAVEGACIÓN ─────────────────────────────────────────

\begin{entrada}{2025-XX-XX}{Intento de SLAM manual con slam\_toolbox — limitación del entorno}
  \problema{Configurar \texttt{slam\_toolbox} manualmente para generar un mapa del entorno simulado.}
  \causa{El \texttt{frame\_id} del scan publicado por el LiDAR en Ignition Gazebo es \texttt{turtlebot4/rplidar\_link/rplidar} (con namespace del robot), pero el árbol TF estático solo contiene \texttt{rplidar\_link} (sin namespace). Esta discrepancia hace que \texttt{slam\_toolbox} descarte todos los scans. Adicionalmente, la simulación corre al $\approx 20\%$ de la velocidad real, lo que agrava los problemas de sincronización de timestamps entre TF y scans.}
  \solucion{Se descartó el SLAM manual en simulación. Se adoptó el mapa pregenerado del almacén incluido en \texttt{turtlebot4\_navigation} (\texttt{warehouse.yaml} + \texttt{warehouse.pgm}) como base para los experimentos de navegación.}
  \secciondestino{5.3 — Metodología experimental (limitaciones del entorno)}
  \nota{El SLAM en el robot real no tiene este problema: el hardware publica los TF con timestamps correctos y sin problemas de namespace. El SLAM real se realizará en la fase de pruebas con el robot físico.}
  \resultado{Decisión: usar mapa pregenerado en simulación y SLAM real en robot físico. Ver §\,7.2 (Líneas futuras).}
\end{entrada}


\begin{entrada}{2025-XX-XX}{Puesta en marcha de Nav2 con mapa pregenerado}
  \problema{Configurar Nav2 para navegación autónoma en el entorno simulado usando el mapa del almacén.}
  \solucion{Lanzamiento con:
\begin{lstlisting}[language=bash]
ros2 launch turtlebot4_ignition_bringup turtlebot4_ignition.launch.py \
  nav2:=true localization:=true
ros2 launch turtlebot4_viz view_robot.launch.py
\end{lstlisting}
  Nav2 arranca con AMCL para localización y NavFn (Dijkstra) como planificador global por defecto.}
  \secciondestino{5.1 — Introducción a la implementación}
  \nota{Es necesario hacer undock del robot antes de poder moverlo, ya sea con \texttt{ros2 action send\_goal /undock irobot\_create\_msgs/action/Undock \{\}} o desde el visor de Gazebo.}
\end{entrada}


\begin{entrada}{2025-XX-XX}{AMCL — localización por partículas}
  \problema{Al arrancar Nav2, el robot no aparece localizado en el mapa y los servidores de planificación dan error de TF (\textit{``frame map does not exist''}).}
  \causa{AMCL (Adaptive Monte Carlo Localization) necesita una pose inicial aproximada para inicializar las partículas. Sin ella, las hipótesis de posición están dispersas por todo el mapa y la transformada \texttt{map~$\rightarrow$~odom} no se publica.}
  \solucion{Usar el botón \textit{2D Pose Estimate} en RViz para indicar la posición y orientación aproximada del robot en el mapa. Las partículas (visibles como puntos verdes) convergen y Nav2 queda operativo.}
  \secciondestino{3.2 — Marco teórico (AMCL)}
  \nota{La convergencia de AMCL es menos precisa en zonas con geometría repetitiva (pasillos similares) y se ve afectada por la baja velocidad de simulación ($\approx 20\%$). En el robot real este problema es menos pronunciado.}
\end{entrada}


\begin{entrada}{2025-XX-XX}{Nav2 operativo — planificador Dijkstra (baseline)}
  \solucion{Nav2 calcula y ejecuta trayectorias autónomas usando \texttt{NavfnPlanner} con Dijkstra (\texttt{use\_astar: false}). La trayectoria se visualiza en RViz como una línea rosa. El robot navega autónomamente hasta el destino.}
  \secciondestino{5.2 — Desarrollo de algoritmos (planificador baseline)}
  \resultado{Este es el planificador de referencia. Métricas a medir: longitud de trayectoria (m), tiempo de planificación (ms) y suavidad (cambios de dirección). Ver §\,6.3.}
  \nota{Configuración en \texttt{nav2.yaml}: \texttt{plugin: "nav2\_navfn\_planner/NavfnPlanner"}, \texttt{use\_astar: false}.}
\end{entrada}

\begin{entrada}{2025-XX-XX}{Launch file personalizado sim.launch.py}
  \solucion{Se crea \texttt{sim.launch.py} en el paquete \texttt{turtlebot4\_cpp\_slam\_nav} que lanza Gazebo, Nav2, RViz y automatiza la pose inicial. Acepta argumentos \texttt{planner} (dijkstra/astar), \texttt{world} (warehouse/depot/maze) y \texttt{launch\_rviz} (true/false). El planificador se selecciona cargando el yaml correspondiente (\texttt{nav2\_dijkstra.yaml}, \texttt{nav2\_astar.yaml}, etc.).}
\begin{lstlisting}[language=bash]
# Lanzar con Dijkstra (por defecto)

# Lanzar con A*
ros2 launch turtlebot4_cpp_slam_nav sim.launch.py planner:=astar

# Sin suavizado
ros2 launch turtlebot4_cpp_slam_nav sim.launch.py smooth:=false

# Sin RViz
ros2 launch turtlebot4_cpp_slam_nav sim.launch.py launch_rviz:=false
\end{lstlisting}
  \secciondestino{5.3 — Metodología experimental}
  \nota{El argumento se llama \texttt{launch\_rviz} en lugar de \texttt{rviz} para evitar conflicto con el argumento homónimo del launch de Gazebo, que lo sobreescribía a \texttt{false}.}
\end{entrada}


\begin{entrada}{2025-XX-XX}{Automatización de pose inicial, undock y giro en el launch file}
  \problema{Para reproducibilidad de los experimentos, el robot debe partir siempre de la misma posición y orientación sin intervención manual.}
  \solucion{Se añaden cuatro \texttt{TimerAction} al launch file con los siguientes tiempos (absolutos desde el arranque):
  \begin{itemize}
    \item \texttt{t=60s} — publicar pose inicial en \texttt{/initialpose} (AMCL)
    \item \texttt{t=90s} — desactivar límite de backup: \texttt{ros2 param set /motion\_control safety\_override backup\_only}
    \item \texttt{t=95s} — undock: \texttt{ros2 action send\_goal /undock}
    \item \texttt{t=110s} — giro de $\pi$ radianes: \texttt{ros2 action send\_goal /spin nav2\_msgs/action/Spin \{"target\_yaw": 3.14159\}}
    \item \texttt{t=135s} — nodo de métricas
  \end{itemize}}
  \secciondestino{5.3 — Metodología experimental}
  \nota{Los tiempos son largos debido a la simulación al 20\% de velocidad real. El \texttt{safety\_override} es necesario porque el undock falla intermitentemente por el límite de backup del Create3 en simulación.}
\end{entrada}


\begin{entrada}{2025-XX-XX}{Problema: costmap local inflado alrededor del robot}
  \problema{El costmap local mostraba una zona rosa/morada enorme alrededor del robot, haciendo que el spin y otros comportamientos abortaran con \textit{``Collision Ahead''}.}
  \causa{El parámetro \texttt{obstacle\_min\_range: 0.0} hacía que el costmap marcara como obstáculo cualquier lectura del LiDAR, incluyendo las muy cercanas al robot (partes del propio robot, dock).}
  \solucion{Cambiar en el yaml de Nav2:
\begin{lstlisting}[language=yaml]
voxel_layer:
  scan:
    obstacle_min_range: 0.2
    raytrace_min_range: 0.2
obstacle_layer:
  scan:
    obstacle_min_range: 0.2
    raytrace_min_range: 0.2
\end{lstlisting}}
  \secciondestino{5.3 — Metodología experimental (incidencias)}
  \resultado{El costmap local quedó limpio alrededor del robot, permitiendo el spin y mejorando la navegación local.}
\end{entrada}


\begin{entrada}{2025-XX-XX}{Problema: undock intermitente en simulación}
  \problema{El undock automático falla en la mayoría de ejecuciones con \textit{``Reached backup limit''}, aunque ocasionalmente tiene éxito.}
  \causa{El Create3 en simulación tiene un límite de distancia de retroceso durante el undock. El parámetro \texttt{safety\_override} debería desactivarlo, pero el nodo \texttt{/motion\_control} no siempre está disponible cuando se intenta setear.}
  \solucion{Ajustar los tiempos del \texttt{TimerAction} para asegurar que \texttt{/motion\_control} está disponible antes de intentar el \texttt{safety\_override}. El goal del undock termina con estado \texttt{CANCELED} pero \texttt{is\_docked: false} — el robot sí se separa físicamente del dock.}
  \secciondestino{5.3 — Metodología experimental (limitaciones del entorno)}
  \nota{El plugin de Ignition del Create3 no implementa correctamente la detección de ``undock completado'' — en el robot real el undock siempre termina con \texttt{SUCCEEDED} usando sensores IR físicos. No tiene solución en simulación con Humble. El comportamiento es funcional aunque el status sea incorrecto.}
\end{entrada}


\begin{entrada}{2025-XX-XX}{Yamls personalizados de Nav2 por algoritmo}
  \solucion{Se crean yamls completos (copia del oficial) por algoritmo en \texttt{config/}:
  \begin{itemize}
    \item \texttt{nav2\_dijkstra.yaml} — \texttt{use\_astar: false}
    \item \texttt{nav2\_astar.yaml} — \texttt{use\_astar: true}
    \item \texttt{nav2\_dijkstra\_nosmooth.yaml} — Dijkstra sin suavizado (\texttt{do\_refinement: False})
    \item \texttt{nav2\_astar\_nosmooth.yaml} — A* sin suavizado
  \end{itemize}
  El launch file selecciona el yaml mediante \texttt{PythonExpression} combinando los argumentos \texttt{planner} y \texttt{smooth}.}
  \secciondestino{5.2 — Desarrollo de algoritmos}
  \nota{Se intentó usar yaml mínimo con \texttt{RewrittenYaml} para fusionar con el oficial, pero \texttt{params\_file} reemplaza completamente el yaml oficial en lugar de fusionarlo. Solución adoptada: yaml completo por algoritmo.}
\end{entrada}

\begin{entrada}{2025-XX-XX}{Decisión de diseño: nodo estático vs.\ nodo dinámico}
  \solucion{Se diseñan dos nodos de medición con objetivos complementarios:
  \begin{itemize}
    \item \textbf{Nodo estático (\texttt{data\_measure.cpp}):} mide métricas sobre la trayectoria \textit{planificada}. Cancela la navegación inmediatamente tras recibir el plan. Se usa en el portátil (simulación al 20\%) donde el robot no navega correctamente. Procesa 4 goals distintos por ejecución.
    \item \textbf{Nodo dinámico (\texttt{data\_measure\_dynamic.cpp}):} mide métricas sobre la trayectoria \textit{ejecutada} en tiempo real. El robot navega hasta el destino. Se usa en el PC de alto rendimiento donde la simulación va más fluida. Procesa 1 goal real por ejecución.
  \end{itemize}}
  \secciondestino{5.3 — Metodología experimental}
\end{entrada}


\begin{entrada}{2025-XX-XX}{Nodo estático: data\_measure.cpp}
  \solucion{El nodo actúa como action client de Nav2, envía goals programáticamente, recibe la trayectoria planificada por \texttt{/plan} y calcula:
  \begin{itemize}
    \item \textbf{Tiempo de planificación} — diferencia entre envío del goal y recepción del plan (ms)
    \item \textbf{Longitud total} — suma de distancias entre puntos consecutivos (m)
    \item \textbf{Distancia en línea recta} — distancia euclídea origen-destino (m)
    \item \textbf{Ratio longitud} — longitud real / línea recta (1.0 = trayectoria perfecta)
    \item \textbf{Suavidad} — suma de cambios de ángulo entre segmentos (rad)
    \item \textbf{Curvatura media} — suavidad / longitud (rad/m)
    \item \textbf{Número de waypoints} — puntos intermedios del plan
    \item \textbf{ETA aproximado} — longitud / 0.3 m/s (velocidad máxima del Create3)
  \end{itemize}
  Goals definitivos: \texttt{warmup} $(-1, 0)$, \texttt{goal1\_recto} $(-6, 0)$, \texttt{goal2\_obstaculos} $(2, -5)$, \texttt{goal3\_lejos} $(-7, 13)$, \texttt{goal4\_atras} $(10, 7)$.}
  \secciondestino{5.2 — Desarrollo de algoritmos}
  \nota{El CSV se nombra dinámicamente: \texttt{metrics\_dijkstra.csv}, \texttt{metrics\_astar\_nosmooth.csv}, etc. Se elimina al arrancar el nodo para evitar acumulación de ejecuciones anteriores.}
\end{entrada}


\begin{entrada}{2025-XX-XX}{Decisión: goal de calentamiento para métricas consistentes}
  \problema{El primer goal de cada ejecución tardaba significativamente más en planificarse ($\approx 36$\,ms vs.\ $\approx 17$\,ms) porque Nav2 inicializa cachés internas en la primera planificación.}
  \solucion{Se añade un goal de calentamiento (\texttt{"warmup"}) al principio del array. Este goal se planifica y cancela igual que los demás, pero sus métricas no se guardan en el CSV.}
  \secciondestino{5.3 — Metodología experimental}
  \resultado{El tiempo de planificación del primer goal real pasó de $\approx 36$\,ms a $\approx 17$\,ms, consistente con el resto.}
\end{entrada}


\begin{entrada}{2025-XX-XX}{Problema: ETA siempre 0.0 desde el feedback de Nav2}
  \problema{El campo \texttt{eta\_s} aparecía siempre a 0.0 en el CSV.}
  \causa{El plan llega muy rápido ($\approx 17$\,ms) pero el primer feedback de Nav2 con el ETA tarda más. Al cancelar inmediatamente tras recibir el plan, el \texttt{feedback\_callback} nunca se ejecuta.}
  \solucion{ETA calculado como aproximación: $\text{ETA} = \text{longitud} / 0.3$\,m/s. Consistente entre algoritmos y suficiente para la comparativa en simulación.}
  \secciondestino{5.3 — Metodología experimental (decisiones técnicas)}
  \nota{En el nodo dinámico el ETA real sí estará disponible desde el feedback de Nav2, ya que el robot navega hasta el destino sin cancelar.}
\end{entrada}


\begin{entrada}{2025-XX-XX}{Problema: spin ignorado por motion\_control}

  \secciondestino{5.3 — Metodología experimental (incidencias)}
\end{entrada}

\begin{entrada}{2025-XX-XX}{Problema: multicast no disponible en Docker}
  \problema{La consola mostraba continuamente \textit{``Exception sending a multicast message: No such device''} al correr el nodo de métricas dentro del contenedor Docker.}
  \causa{Docker con \texttt{--network host} no soporta multicast en algunos entornos de red. ROS2 usa multicast para el descubrimiento de nodos (DDS).}
  \solucion{Añadir \texttt{export ROS\_LOCALHOST\_ONLY=1} al \texttt{.bashrc} del contenedor y al \texttt{Dockerfile}. Fuerza a ROS2 a usar solo unicast en localhost, suficiente cuando todos los nodos corren en el mismo PC.}
  \secciondestino{5.3 — Metodología experimental (incidencias)}
  \nota{No afecta al funcionamiento — ROS2 cae back a unicast automáticamente. La solución evita el ruido en la consola.}
\end{entrada}
